## WebSocket Architecture

### Backend (FastAPI Python)

```python
# backend/ws_server.py
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
import json
import asyncio

app = FastAPI()

class NegotiationBroadcaster:
    """Broadcast negotiation events to all connected dashboards."""
    def __init__(self):
        self.connections: list[WebSocket] = []
    
    async def connect(self, ws: WebSocket):
        await ws.accept()
        self.connections.append(ws)
    
    def disconnect(self, ws: WebSocket):
        self.connections.remove(ws)
    
    async def broadcast(self, event: dict):
        """Send negotiation event to all connected dashboards."""
        data = json.dumps(event)
        for conn in self.connections:
            await conn.send_text(data)

broadcaster = NegotiationBroadcaster()

@app.websocket("/ws/negotiation/{session_id}")
async def negotiation_stream(websocket: WebSocket, session_id: str):
    await broadcaster.connect(websocket)
    try:
        while True:
            # Receive commands from dashboard (e.g., start negotiation)
            data = await websocket.receive_text()
            command = json.loads(data)
            if command["type"] == "START_NEGOTIATION":
                await start_negotiation(session_id, command, broadcaster)
    except WebSocketDisconnect:
        broadcaster.disconnect(websocket)

# Event types the backend emits:
NEGOTIATION_EVENTS = {
    "NEGOTIATION_STARTED": {"buyer": str, "seller": str, "resource": str},
    "OFFER_MADE":          {"round": int, "agent": str, "price": int, "timestamp": str},
    "COUNTER_OFFER":       {"round": int, "agent": str, "price": int, "timestamp": str},
    "DEAL_AGREED":         {"round": int, "price": int, "deal_hash": str},
    "DEAL_REJECTED":       {"reason": str},
    "PAYMENT_REQUIRED":    {"amount": int, "network": str, "asset": str},
    "PAYMENT_SUBMITTED":   {"tx_hash": str},
    "PAYMENT_SETTLED":     {"tx_hash": str, "block": int},
    "ATTESTATION_WRITTEN": {"deal_hash": str, "tx_hash": str},
    "AGENT_THINKING":      {"agent": str, "reasoning": str, "strategy": str},
    "MCP_TOOL_DISCOVERED": {"tool_name": str, "endpoint": str, "price": str},
    "ERROR":               {"code": str, "message": str, "retrying": bool},
}
```

### Frontend (Next.js WebSocket Hook)

```typescript
// hooks/useNegotiationStream.ts
"use client";
import { useEffect, useRef, useState, useCallback } from "react";

export type NegotiationEvent = {
  type: string;
  timestamp: string;
  data: Record<string, any>;
};

export function useNegotiationStream(sessionId: string) {
  const [events, setEvents] = useState<NegotiationEvent[]>([]);
  const [connected, setConnected] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    const ws = new WebSocket(
      `ws://localhost:8000/ws/negotiation/${sessionId}`
    );

    ws.onopen = () => setConnected(true);
    ws.onclose = () => setConnected(false);
    ws.onmessage = (event) => {
      const parsed: NegotiationEvent = JSON.parse(event.data);
      setEvents((prev) => [...prev, parsed]);
    };

    wsRef.current = ws;
    return () => ws.close();
  }, [sessionId]);

  const sendCommand = useCallback((command: Record<string, any>) => {
    wsRef.current?.send(JSON.stringify(command));
  }, []);

  return { events, connected, sendCommand };
}
```

## D. Price Convergence Chart (Recharts v3)

```tsx
// components/charts/price-convergence.tsx
"use client";

import { LineChart, Line, XAxis, YAxis, CartesianGrid, ReferenceLine } from "recharts";
import { ChartContainer, ChartTooltip, ChartTooltipContent } from "@/components/ui/chart";
import { useNegotiationStream } from "@/hooks/useNegotiationStream";

const chartConfig = {
  buyerOffer: { label: "Buyer Offer", color: "var(--chart-1)" },
  sellerAsk:  { label: "Seller Ask",  color: "var(--chart-2)" },
  agreedPrice: { label: "Agreed Price", color: "var(--chart-3)" },
};

export function PriceConvergenceChart({ sessionId }: { sessionId: string }) {
  const { events } = useNegotiationStream(sessionId);

  // Transform events into chart data points
  const chartData = events
    .filter((e) => ["OFFER_MADE", "COUNTER_OFFER", "DEAL_AGREED"].includes(e.type))
    .reduce((acc, event) => {
      const round = event.data.round;
      const existing = acc.find((d) => d.round === round) || { round };
      if (event.data.agent === "buyer") existing.buyerOffer = event.data.price / 1e6;
      if (event.data.agent === "seller") existing.sellerAsk = event.data.price / 1e6;
      if (event.type === "DEAL_AGREED") existing.agreedPrice = event.data.price / 1e6;

      const idx = acc.findIndex((d) => d.round === round);
      if (idx >= 0) acc[idx] = existing;
      else acc.push(existing);
      return acc;
    }, [] as any[]);

  return (
    <ChartContainer config={chartConfig} className="min-h-[300px] w-full">
      <LineChart data={chartData} accessibilityLayer>
        <CartesianGrid vertical={false} />
        <XAxis dataKey="round" label={{ value: "Round", position: "bottom" }} />
        <YAxis label={{ value: "Price (USDC)", angle: -90 }} />
        <ChartTooltip content={<ChartTooltipContent />} />

        <Line
          type="monotone"
          dataKey="buyerOffer"
          stroke="var(--color-buyerOffer)"
          strokeWidth={2}
          dot={{ r: 4 }}
          isAnimationActive={true}
          animationDuration={300}
        />
        <Line
          type="monotone"
          dataKey="sellerAsk"
          stroke="var(--color-sellerAsk)"
          strokeWidth={2}
          dot={{ r: 4 }}
          isAnimationActive={true}
          animationDuration={300}
        />

        {chartData.some((d) => d.agreedPrice) && (
          <ReferenceLine
            y={chartData.find((d) => d.agreedPrice)?.agreedPrice}
            stroke="var(--color-agreedPrice)"
            strokeDasharray="3 3"
            label="Deal"
          />
        )}
      </LineChart>
    </ChartContainer>
  );
}
```

**Source**: [shadcn/ui Chart docs](https://ui.shadcn.com/docs/components/chart)

## E. MCP Discovery Panel

```tsx
// components/mcp/discovery-panel.tsx
"use client";

import { motion, AnimatePresence } from "framer-motion";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Zap, DollarSign, Globe } from "lucide-react";
import { useNegotiationStream } from "@/hooks/useNegotiationStream";

export function McpDiscoveryPanel({ sessionId }: { sessionId: string }) {
  const { events } = useNegotiationStream(sessionId);

  const tools = events
    .filter((e) => e.type === "MCP_TOOL_DISCOVERED")
    .map((e) => e.data);

  return (
    <Card>
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <Globe className="h-4 w-4" />
          MCP Tool Discovery
          <Badge variant="outline" className="ml-auto">
            {tools.length} found
          </Badge>
        </CardTitle>
      </CardHeader>
      <CardContent>
        <AnimatePresence>
          {tools.map((tool, i) => (
            <motion.div
              key={tool.endpoint}
              initial={{ opacity: 0, x: -20, height: 0 }}
              animate={{ opacity: 1, x: 0, height: "auto" }}
              exit={{ opacity: 0, x: 20 }}
              transition={{ duration: 0.3, delay: i * 0.1 }}
              className="flex items-center justify-between p-3 border rounded-lg mb-2"
            >
              <div className="flex items-center gap-3">
                <motion.div
                  initial={{ scale: 0 }}
                  animate={{ scale: 1 }}
                  transition={{ type: "spring", stiffness: 500, delay: i * 0.1 + 0.2 }}
                >
                  <Zap className="h-4 w-4 text-yellow-500" />
                </motion.div>
                <div>
                  <p className="font-medium text-sm">{tool.tool_name}</p>
                  <p className="text-xs text-muted-foreground truncate max-w-[200px]">
                    {tool.endpoint}
                  </p>
                </div>
              </div>
              <Badge variant="secondary" className="flex items-center gap-1">
                <DollarSign className="h-3 w-3" />
                {tool.price}
              </Badge>
            </motion.div>
          ))}
        </AnimatePresence>

        {tools.length === 0 && (
          <div className="flex items-center justify-center h-24 text-muted-foreground">
            <motion.div
              animate={{ opacity: [0.3, 1, 0.3] }}
              transition={{ repeat: Infinity, duration: 2 }}
            >
              Scanning for x402-gated services...
            </motion.div>
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```

## Key Implementation Patterns

### On-Chain Data via wagmi Hooks

```tsx
// hooks/useKiteBalance.ts
import { useBalance } from "wagmi";
import { kiteTestnet, KITE_TEST_USDT } from "@/lib/chains/kite";

export function useKiteBalance(address: `0x${string}`) {
  // Native KITE balance
  const kite = useBalance({ address, chainId: kiteTestnet.id });
  
  // USDT token balance
  const usdt = useBalance({
    address,
    chainId: kiteTestnet.id,
    token: KITE_TEST_USDT,
  });

  return { kite, usdt };
}
```

```tsx
// hooks/useDealRecordEvents.ts
import { useWatchContractEvent } from "wagmi";
import { DEAL_RECORD_ABI } from "@/lib/contracts/deal-record-abi";

export function useDealRecordEvents(contractAddress: `0x${string}`) {
  useWatchContractEvent({
    address: contractAddress,
    abi: DEAL_RECORD_ABI,
    eventName: "DealRecorded",
    onLogs(logs) {
      // Update attestation feed in real-time
      console.log("New deal attestation:", logs);
    },
  });
}
```

### MCP Discovery with use-mcp

```tsx
// components/mcp/discovery-panel.tsx (alternative with Cloudflare use-mcp)
import { useMcp } from "use-mcp/react";

export function McpDiscoveryLive() {
  const { state, tools, callTool } = useMcp({
    url: "https://mcp.prod.gokite.ai/",
  });

  return (
    <div>
      {state === "connecting" && <p>Connecting to Kite MCP...</p>}
      {state === "ready" && tools.map((tool) => (
        <ToolCard key={tool.name} tool={tool} />
      ))}
    </div>
  );
}
```

## Demo Flow (What Judges See)

### Scene 1: Discovery (10s)
- MCP Discovery Panel animates — tools appear one by one with spring animation
- Agent Identity Cards show ERC-8004 registration status with green checkmarks
- Facilitator Status shows "Pieverse: Connected" with green dot

### Scene 2: Negotiation (30s)
- Price Convergence Chart live-updates — two lines converging from opposite sides
- Timeline adds rounds in real-time with offer/counter-offer badges
- Agent Decision Log streams LLM reasoning: "Applying time-dependent concession strategy. Opponent's concession rate is 0.02/round, suggesting a firm reservation price..."
- Final round: lines converge, "DEAL AGREED" badge appears with deal_hash

### Scene 3: Payment (10s)
- Activity Feed shows: "Payment Required → Payment Submitted → Payment Settled"
- Wallet Balance updates in real-time (USDT decreases by negotiated amount)
- Tx hash appears as a clickable link to KiteScan

### Scene 4: Attestation (10s)
- Attestation Feed shows new DealRecord event from on-chain
- Deal History Table adds a new row with expandable negotiation rounds
- Reputation scores update via ERC-8004 Reputation Registry

Total demo time: ~60 seconds of animated, real-time data flowing through the entire NegotiatorGrid pipeline.
